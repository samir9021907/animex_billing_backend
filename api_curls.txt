# ------------------------------------------------------------------------------
# 2. CLIENT MODULE (/client)
# ------------------------------------------------------------------------------

# 2.1 Create Client (No Token Required)
curl --location 'https://animex-pharma-backend.onrender.com/client/clients' \
--header 'Content-Type: application/json' \
--data '{
    "name": "Apollo Pharmacy",
    "email": "contact@apollopharmacy.com",
    "phone": "+919876543210",
    "address": "123 Main Medical Street",
    "city": "Mumbai",
    "status": "active"
}'

# 2.2 Get All Clients (No Token Required)
curl --location 'https://animex-pharma-backend.onrender.com/client/clients'

# 2.3 Get Client By ID (Requires Authorization Header)
curl --location 'https://animex-pharma-backend.onrender.com/client/clients/CLIENT_ID' \
--header 'Authorization: Bearer YOUR_JWT_TOKEN'

# 2.4 Update Client (Requires Authorization Header)
curl --location --request PUT 'https://animex-pharma-backend.onrender.com/client/clients/CLIENT_ID' \
--header 'Content-Type: application/json' \
--header 'Authorization: Bearer YOUR_JWT_TOKEN' \
--data '{
    "name": "Apollo Pharmacy Pvt Ltd",
    "phone": "+919876543211",
    "status": "active"
}'

# 2.5 Delete Client (Requires Authorization Header)
curl --location --request DELETE 'https://animex-pharma-backend.onrender.com/client/clients/CLIENT_ID' \
--header 'Authorization: Bearer YOUR_JWT_TOKEN'
